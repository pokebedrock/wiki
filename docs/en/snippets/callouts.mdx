export const InfoCallout = () => (
  <div className="callout callout--note">
    <strong>Heads up:</strong> Components exported from <code>docs/&lt;locale&gt;/snippets/</code>
    are auto-registered, so you don’t need to import helpers such as <code>ShellBlock</code>
    manually.
  </div>
);

export const WarningCallout = () => (
  <div className="callout callout--warning">
    <strong>Warning:</strong> Double-check the in-game release notes before updating this doc.
  </div>
);

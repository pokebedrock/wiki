export const InfoCallout = () => (
  <div className="callout callout--note">
    <strong>Heads up:</strong> Run `npm run ci` and review the generated search artifacts before opening or updating a PR.
  </div>
);

export const WarningCallout = () => (
  <div className="callout callout--warning">
    <strong>Warning:</strong> When a doc depends on automation or generated content, verify the current workflow behavior before publishing changes.
  </div>
);

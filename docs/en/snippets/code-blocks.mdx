const shellTemplate = `#!/bin/bash
set -euo pipefail

`;

const earlyReturnSnippet = [
  "function ensurePlayerCanTrade(player) {",
  "  if (!player?.inventory) {",
  "    return false;",
  "  }",
  "",
  "  if (player.inventory.slots < 1) {",
  "    return false;",
  "  }",
  "",
  "  return true;",
  "}"
].join("\n");

export const ShellBlock = ({ command }) => (
  <pre>
    <code>{`${shellTemplate}${command}`}</code>
  </pre>
);

export const EarlyReturnExample = () => (
  <pre>
    <code>{earlyReturnSnippet}</code>
  </pre>
);
